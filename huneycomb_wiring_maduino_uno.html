// ===================================================================== // Huneycomb - hive sensor node // Sprint 1: bench test (sensor powered + connected to laptop over USB) // // Boards (auto-detected from Tools > Board): // Maduino LoRa Radio v2.0 -> "Arduino Pro or Pro Mini", // Processor "ATmega328P (3.3V, 8 MHz)" // Elegoo UNO R3 (bench) -> "Arduino Uno" // Serial: 9600 baud // // Wiring (Maduino): // BME280 VCC->3V3 GND->GND SDA->A4 SCL->A5 // DS3231 RTC VCC->3V3 GND->GND SDA->A4 SCL->A5 (same I2C bus) // OSEPP temp S->A0 +->3V3 -->GND // microSD CS->D8 MOSI->D11 MISO->D12 SCK->D13 (optional) // Pins used by the on-board radio: D10 (CS), D9 (RST), D2 (DIO0), // and D3/D4/D5 while the DIO1-3 jumpers are fitted. // // Serial commands (type a letter + Enter): // h help r read now s I2C scan i info f fast mode // p pause c set clock to compile time d dump SD log // ===================================================================== #include #include #include #include // --------------------------------------------------------------------- // Configuration // --------------------------------------------------------------------- #define USE_SD 0 // 1 = log to SD card #define USE_PROBE 1 // 1 = read OSEPP analog temp sensor on A0 #define USE_RTC 1 // 1 = timestamp readings from the RTC #define RFM95_CS 10 // on-board radio chip-select (keep HIGH!) #define RFM95_RST 9 #define PROBE_PIN A0 #define LED_PIN 13 #if defined(ARDUINO_AVR_UNO) // Bench: no radio, HiLetgo logger shield (CS hard-wired to D10, // DS1307 clock). Define FORCE_DS3231 to use the ZS-042 board instead. #define BOARD_NAME "UNO (test bench)" #define SD_CS 10 #ifdef FORCE_DS3231 #define RTC_IS_3231 1 #else #define RTC_IS_3231 0 #endif #else #define BOARD_NAME "Maduino LoRa" #define SD_CS 8 // must NOT be 10 - that's the radio #define RTC_IS_3231 1 // ZS-042 DS3231 module #endif #define INTERVAL_MS 10000UL #define FAST_INTERVAL_MS 2000UL // --------------------------------------------------------------------- #if USE_SD #include const char LOG_FILE[] = "HIVE.CSV"; bool sdOk = false; #endif #if USE_RTC #include #if RTC_IS_3231 RTC_DS3231 rtc; #define RTC_NAME "DS3231" #else RTC_DS1307 rtc; #define RTC_NAME "DS1307 (logger shield)" #endif bool rtcOk = false; #endif Adafruit_BME280 bme; uint8_t bmeAddr = 0; // 0 = not found uint32_t seq = 0; // reading counter (later: packet number) uint32_t lastRead = 0; bool fastMode = false; bool paused = false; char stampBuf[20]; // "YYYY-MM-DD hh:mm:ss" // --------------------------------------------------------------------- // Helpers // --------------------------------------------------------------------- int freeRam() { extern int __heap_start, *__brkval; int v; return (int)&v - (__brkval == 0 ? (int)&__heap_start : (int)__brkval); } // Chip supply voltage measured against the internal 1.1V reference. long readVccMv() { ADMUX = _BV(REFS0) | _BV(MUX3) | _BV(MUX2) | _BV(MUX1); delay(2); ADCSRA |= _BV(ADSC); while (bit_is_set(ADCSRA, ADSC)); long result = ADCL; result |= ADCH << 8; return 1125300L / result; // 1.1 * 1023 * 1000 } void blink(uint8_t n) { for (uint8_t i = 0; i < n; i++) { digitalWrite(LED_PIN, HIGH); delay(60); digitalWrite(LED_PIN, LOW); delay(120); } } // Fills stampBuf with the RTC time, or "" if there's no clock. const char* timestamp() { stampBuf[0] = 0; #if USE_RTC if (rtcOk) { DateTime n = rtc.now(); snprintf(stampBuf, sizeof stampBuf, "%04d-%02d-%02d %02d:%02d:%02d", n.year(), n.month(), n.day(), n.hour(), n.minute(), n.second()); } #endif return stampBuf; } void printHelp() { Serial.print(F("Commands: h=help r=read s=scan i=info f=fast p=pause")); #if USE_RTC Serial.print(F(" c=set-clock")); #endif #if USE_SD Serial.print(F(" d=dump-sd")); #endif Serial.println(); } // --------------------------------------------------------------------- // I2C scan // --------------------------------------------------------------------- uint8_t scanI2C() { uint8_t found = 0; Serial.println(F("I2C scan:")); for (uint8_t a = 1; a < 127; a++) { Wire.beginTransmission(a); if (Wire.endTransmission() == 0) { Serial.print(F(" device at 0x")); if (a < 16) Serial.print('0'); Serial.print(a, HEX); if (a == 0x76 || a == 0x77) Serial.print(F(" <- BME280")); if (a == 0x68) Serial.print(F(" <- RTC (DS3231/DS1307)")); if (a == 0x57) Serial.print(F(" <- EEPROM on ZS-042 clock board")); Serial.println(); found++; } } if (!found) { Serial.println(F(" nothing found. Check: VCC on 3V3, GND, SDA->A4, SCL->A5,")); Serial.println(F(" and that SDA/SCL aren't swapped.")); } return found; } bool initBME() { const uint8_t addrs[] = {0x76, 0x77}; for (uint8_t i = 0; i < 2; i++) { if (bme.begin(addrs[i])) { bmeAddr = addrs[i]; bme.setSampling(Adafruit_BME280::MODE_FORCED, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::SAMPLING_X1, Adafruit_BME280::FILTER_OFF); return true; } } bmeAddr = 0; Serial.print(F("BME280 not found. Sensor ID reported: 0x")); Serial.println(bme.sensorID(), HEX); Serial.println(F(" 0xFF = wrong address/wiring, 0x56-0x58 = BMP280 (no humidity),")); Serial.println(F(" 0x60 = BME280, 0x61 = BME680")); return false; } #if USE_RTC void setClockToCompileTime() { rtc.adjust(DateTime(F(__DATE__), F(__TIME__))); Serial.print(F("Clock set to compile time: ")); Serial.println(timestamp()); } bool initRTC() { Serial.print(F("RTC " RTC_NAME ": ")); if (!rtc.begin()) { Serial.println(F("not found at 0x68")); return false; } rtcOk = true; #if RTC_IS_3231 bool lost = rtc.lostPower(); #else bool lost = !rtc.isrunning(); #endif if (lost) { Serial.println(F("OK, but time was lost (battery?)")); setClockToCompileTime(); } else { Serial.print(F("OK, ")); Serial.println(timestamp()); } return true; } #endif #if USE_SD bool initSD() { Serial.print(F("SD card on CS D")); Serial.print(SD_CS); Serial.print(F(": ")); if (!SD.begin(SD_CS)) { Serial.println(F("FAILED (card inserted? FAT32? CS pin right?)")); return false; } Serial.println(F("OK")); if (!SD.exists(LOG_FILE)) { File f = SD.open(LOG_FILE, FILE_WRITE); if (f) { f.println(F("seq,time,ms,temp_c,rh_pct,press_hpa,probe_raw,rtc_temp_c,vcc_mv")); f.close(); } } return true; } void dumpSD() { if (!sdOk) { Serial.println(F("SD not available")); return; } File f = SD.open(LOG_FILE); if (!f) { Serial.println(F("can't open log")); return; } Serial.println(F("----- HIVE.CSV -----")); while (f.available()) Serial.write(f.read()); f.close(); Serial.println(F("----- end -----")); } #endif // Reads the probe and averages 8 samples. Returns raw 0-1023. int readProbeRaw() { analogRead(PROBE_PIN); // discard first read after mux change long acc = 0; for (uint8_t i = 0; i < 8; i++) { acc += analogRead(PROBE_PIN); delay(2); } return acc / 8; } // The OSEPP board's chip isn't identified yet. Its output voltage at // room temperature tells us which one it is. void probeHint() { #if USE_PROBE long vcc = readVccMv(); float v = readProbeRaw() * (vcc / 1023.0) / 1000.0; Serial.print(F("Probe on A0: ")); Serial.print(v, 3); Serial.print(F(" V -> ")); if (v > 0.9 * vcc / 1000.0) Serial.println(F("near supply: digital sensor (DS18B20?) or S not connected")); else if (v > 0.6 && v < 1.0) Serial.println(F("looks like a TMP36 (0.75 V at 25 C, T = (V-0.5)*100)")); else if (v > 0.1 && v < 0.45) Serial.println(F("looks like an LM35 (0.25 V at 25 C, T = V*100; wants >=4 V supply)")); else Serial.println(F("unclear - check S/+/- wiring")); #endif } void printInfo() { Serial.println(F("--- Huneycomb info ---")); Serial.print(F("Board: ")); Serial.println(F(BOARD_NAME)); Serial.print(F("Free RAM: ")); Serial.print(freeRam()); Serial.println(F(" bytes (of 2048)")); Serial.print(F("Supply: ")); Serial.print(readVccMv()); Serial.println(F(" mV")); Serial.print(F("BME280: ")); if (bmeAddr) { Serial.print(F("0x")); Serial.println(bmeAddr, HEX); } else Serial.println(F("not found")); #if USE_RTC Serial.print(F("Clock: ")); if (rtcOk) Serial.println(timestamp()); else Serial.println(F("not found")); #endif #if USE_SD Serial.print(F("SD: ")); Serial.println(sdOk ? F("OK") : F("not available")); #else Serial.println(F("SD: disabled (USE_SD 0)")); #endif Serial.print(F("Interval: ")); Serial.print((fastMode ? FAST_INTERVAL_MS : INTERVAL_MS) / 1000); Serial.println(paused ? F(" s (paused)") : F(" s")); probeHint(); } // --------------------------------------------------------------------- // One reading: a human line and a CSV line. The CSV line is the format // that goes into the LoRa packet in sprint 3. // --------------------------------------------------------------------- void takeReading() { seq++; float t = NAN, h = NAN, p = NAN, rtcT = NAN; if (bmeAddr) { bme.takeForcedMeasurement(); t = bme.readTemperature(); h = bme.readHumidity(); p = bme.readPressure() / 100.0F; } #if USE_RTC && RTC_IS_3231 if (rtcOk) rtcT = rtc.getTemperature(); // DS3231 has its own thermometer #endif int probeRaw = -1; #if USE_PROBE probeRaw = readProbeRaw(); #endif long vcc = readVccMv(); const char* ts = timestamp(); uint32_t ms = millis(); // human-readable Serial.print(F("#")); Serial.print(seq); if (ts[0]) { Serial.print(' '); Serial.print(ts); } Serial.print(F(" T=")); Serial.print(t, 2); Serial.print(F("C")); Serial.print(F(" RH=")); Serial.print(h, 1); Serial.print(F("%")); Serial.print(F(" P=")); Serial.print(p, 1); Serial.print(F("hPa")); #if USE_PROBE Serial.print(F(" probe=")); Serial.print(probeRaw * (vcc / 1023.0) / 1000.0, 3); Serial.print(F("V")); #endif #if USE_RTC && RTC_IS_3231 Serial.print(F(" rtcT=")); Serial.print(rtcT, 2); Serial.print(F("C")); #endif Serial.print(F(" Vcc=")); Serial.print(vcc); Serial.println(F("mV")); // CSV: seq,time,ms,temp_c,rh_pct,press_hpa,probe_raw,rtc_temp_c,vcc_mv Serial.print(F("CSV,")); Serial.print(seq); Serial.print(','); Serial.print(ts); Serial.print(','); Serial.print(ms); Serial.print(','); Serial.print(t, 2); Serial.print(','); Serial.print(h, 1); Serial.print(','); Serial.print(p, 1); Serial.print(','); Serial.print(probeRaw); Serial.print(','); Serial.print(rtcT, 2); Serial.print(','); Serial.println(vcc); #if USE_SD if (sdOk) { File f = SD.open(LOG_FILE, FILE_WRITE); if (f) { f.print(seq); f.print(','); f.print(ts); f.print(','); f.print(ms); f.print(','); f.print(t, 2); f.print(','); f.print(h, 1); f.print(','); f.print(p, 1); f.print(','); f.print(probeRaw); f.print(','); f.print(rtcT, 2); f.print(','); f.println(vcc); f.close(); } else { Serial.println(F("SD write failed")); } } #endif blink(1); } void handleSerial() { while (Serial.available()) { char c = Serial.read(); switch (c) { case 'h': case '?': printHelp(); break; case 'r': takeReading(); break; case 's': scanI2C(); break; case 'i': printInfo(); break; case 'f': fastMode = !fastMode; Serial.println(fastMode ? F("fast mode ON (2 s)") : F("fast mode OFF")); break; case 'p': paused = !paused; Serial.println(paused ? F("paused") : F("resumed")); break; #if USE_RTC case 'c': if (rtcOk) setClockToCompileTime(); else Serial.println(F("no clock")); break; #endif #if USE_SD case 'd': dumpSD(); break; #endif default: break; } } } // --------------------------------------------------------------------- void setup() { pinMode(LED_PIN, OUTPUT); // The radio shares the SPI bus: keep it deselected so it can't talk // over the SD card. pinMode(RFM95_CS, OUTPUT); digitalWrite(RFM95_CS, HIGH); pinMode(RFM95_RST, OUTPUT); digitalWrite(RFM95_RST, HIGH); Serial.begin(9600); delay(200); Serial.println(); Serial.println(F("=== Huneycomb - sprint 1 bench test ===")); Wire.begin(); scanI2C(); if (initBME()) { Serial.print(F("BME280 OK at 0x")); Serial.println(bmeAddr, HEX); } #if USE_RTC initRTC(); #endif #if USE_SD sdOk = initSD(); #endif printInfo(); printHelp(); blink(bmeAddr ? 2 : 5); // 2 blinks = good, 5 = sensor problem takeReading(); lastRead = millis(); } void loop() { handleSerial(); uint32_t interval = fastMode ? FAST_INTERVAL_MS : INTERVAL_MS; if (!paused && millis() - lastRead >= interval) { lastRead = millis(); takeReading(); } }
